REPORT Z_PUSH_BUTTON25.

*---------------------------------------------------------------------*
* TABLE DECLARATION
*---------------------------------------------------------------------*
TABLES: SSCRFIELDS.

*---------------------------------------------------------------------*
* PARAMETERS
*---------------------------------------------------------------------*
PARAMETERS: N1 TYPE I.

*---------------------------------------------------------------------*
* SELECTION SCREEN
*---------------------------------------------------------------------*
SELECTION-SCREEN SKIP 2.

SELECTION-SCREEN PUSHBUTTON /10(10) PUSH1 USER-COMMAND FCT1.
SELECTION-SCREEN PUSHBUTTON 25(10) PUSH2 USER-COMMAND FCT2.
SELECTION-SCREEN PUSHBUTTON 40(10) PUSH3 USER-COMMAND FCT3.
SELECTION-SCREEN PUSHBUTTON 55(10) PUSH4 USER-COMMAND FCT4.

*---------------------------------------------------------------------*
* INITIALIZATION
*---------------------------------------------------------------------*
INITIALIZATION.
  PUSH1 = 'DISPLAY'.
  PUSH2 = 'RESET'.
  PUSH3 = 'EXPORT'.
  PUSH4 = 'EXIT'.

*---------------------------------------------------------------------*
* HANDLE PUSHBUTTON ACTIONS
*---------------------------------------------------------------------*
AT SELECTION-SCREEN.
  CASE SSCRFIELDS-UCOMM.

*---------------------------------------------------------------------*
* DISPLAY BUTTON
*---------------------------------------------------------------------*
    WHEN 'FCT1'.
      WRITE: / '============================================='.
      WRITE: / 'DISPLAY BUTTON CLICKED'.
      WRITE: / '============================================='.
      WRITE: / 'VALUE OF N1 : ', N1.
      WRITE: / 'DATE        : ', SY-DATUM.
      WRITE: / 'TIME        : ', SY-UZEIT.
      WRITE: / 'USER        : ', SY-UNAME.

      IF N1 MOD 2 = 0.
        WRITE: / 'NUMBER IS EVEN'.
      ELSE.
        WRITE: / 'NUMBER IS ODD'.
      ENDIF.
      WRITE: / '============================================='.

*---------------------------------------------------------------------*
* RESET BUTTON
*---------------------------------------------------------------------*
    WHEN 'FCT2'.
      CLEAR N1.
      WRITE: / '============================================='.
      WRITE: / 'RESET BUTTON CLICKED'.
      WRITE: / '============================================='.
      WRITE: / 'N1 VALUE CLEARED'.
      WRITE: / '============================================='.

*---------------------------------------------------------------------*
* EXPORT BUTTON
*---------------------------------------------------------------------*
    WHEN 'FCT3'.
      WRITE: / '============================================='.
      WRITE: / 'EXPORT BUTTON CLICKED'.
      WRITE: / '============================================='.
      WRITE: / 'DATA EXPORTED SUCCESSFULLY'.
      WRITE: / 'EXPORTED VALUE : ', N1.
      WRITE: / 'EXPORT TIME    : ', SY-UZEIT.
      WRITE: / '============================================='.

*---------------------------------------------------------------------*
* EXIT BUTTON
*---------------------------------------------------------------------*
    WHEN 'FCT4'.
      LEAVE PROGRAM.

  ENDCASE.

*---------------------------------------------------------------------*
* START-OF-SELECTION
*---------------------------------------------------------------------*
START-OF-SELECTION.
  WRITE: / '============================================='.
  WRITE: / 'PROGRAM EXECUTION COMPLETED'.
  WRITE: / '============================================='.
  WRITE: / 'FINAL VALUE OF N1:', N1.
  WRITE: / '============================================='.
