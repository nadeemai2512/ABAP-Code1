        MARC,
        SSCRFIELDS.

* VARIABLES
*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
SELECT-OPTIONS: S_MATNR FOR F_MATNR.

*---------------------------------------------------------------------*

SELECTION-SCREEN PUSHBUTTON 62(10) BUTTON2 USER-COMMAND FT2.

* TYPE DECLARATION
*---------------------------------------------------------------------*
TYPES: BEGIN OF TY_MAT,

         MATNR TYPE MARA-MATNR,
         ERNAM TYPE MARA-ERNAM,
         MTART TYPE MARA-MTART,
         MBRSH TYPE MARA-MBRSH,
         WERKS TYPE MARC-WERKS,
         EKGRP TYPE MARC-EKGRP,
         BSTMI TYPE MARC-BSTMI,

*---------------------------------------------------------------------*

      WA_MAT TYPE TY_MAT.

*---------------------------------------------------------------------*
* FIELD CATALOG
*---------------------------------------------------------------------*
DATA: IT_FCAT TYPE SLIS_T_FIELDCAT_ALV,
      WA_FCAT TYPE SLIS_FIELDCAT_ALV.

*---------------------------------------------------------------------*
* LAYOUT
*---------------------------------------------------------------------*
DATA: WA_LAYOUT TYPE SLIS_LAYOUT_ALV.

*---------------------------------------------------------------------*
* INITIALIZATION
*---------------------------------------------------------------------*
INITIALIZATION.

  BUTTON1 = 'EXECUTE'.
  BUTTON2 = 'EXIT'.

*---------------------------------------------------------------------*
* SELECTION SCREEN VALIDATION
*---------------------------------------------------------------------*
AT SELECTION-SCREEN.

  IF SSCRFIELDS-UCOMM = 'FT2'.

    LEAVE PROGRAM.

  ELSEIF SSCRFIELDS-UCOMM = 'FT1'.

    SSCRFIELDS-UCOMM = 'ONLI'.

  ENDIF.

*---------------------------------------------------------------------*
* START-OF-SELECTION
*---------------------------------------------------------------------*
START-OF-SELECTION.

  PERFORM FETCH_DATA.
  PERFORM BUILD_FCAT.
  PERFORM DISPLAY_ALV.

*---------------------------------------------------------------------*
* FETCH DATA
*---------------------------------------------------------------------*
FORM FETCH_DATA.

  SELECT A~MATNR
         A~ERNAM
         A~MTART
         A~MBRSH
         B~WERKS
         B~EKGRP
         B~BSTMI

    INTO TABLE IT_MAT

    FROM MARA AS A

    INNER JOIN MARC AS B
    ON A~MATNR = B~MATNR

    WHERE A~MATNR IN S_MATNR.

  IF SY-SUBRC <> 0.

    MESSAGE 'NO DATA FOUND' TYPE 'I'.

    STOP.

  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
* DISPLAY ALV
*---------------------------------------------------------------------*
FORM DISPLAY_ALV.

  WA_LAYOUT-ZEBRA = 'X'.
  WA_LAYOUT-COLWIDTH_OPTIMIZE = 'X'.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'

    EXPORTING
      I_CALLBACK_PROGRAM     = SY-REPID
      I_CALLBACK_TOP_OF_PAGE = 'TOP_OF_PAGE'
      IS_LAYOUT              = WA_LAYOUT
      IT_FIELDCAT            = IT_FCAT
      I_GRID_TITLE           = 'MATERIAL MASTER DATA'

    TABLES
      T_OUTTAB               = IT_MAT

    EXCEPTIONS
      PROGRAM_ERROR          = 1
      OTHERS                 = 2.

  IF SY-SUBRC <> 0.

    MESSAGE 'ERROR IN ALV DISPLAY' TYPE 'E'.

  ENDIF.

ENDFORM.
