REPORT Z_INTERNALTABLE.
* EXAMPLE 1: EMPLOYEE
*---------------------------------------------------------------------*
WRITE: / 'INTERNAL TABLE - EMPLOYEE'.
ULINE.

DATA: BEGIN OF rec OCCURS 0,
        eno   TYPE n LENGTH 4,
        ename TYPE c LENGTH 20,
        esal  TYPE c LENGTH 6,
      END OF rec.

rec-eno   = 1001.
rec-ename = 'MOHAN'.
rec-esal  = '712634'.
APPEND rec.

rec-eno   = 1002.
rec-ename = 'RAVI'.
rec-esal  = '500000'.
APPEND rec.

LOOP AT rec.
  WRITE: / rec-eno COLOR 2 INTENSIFIED,
         rec-ename COLOR 3 INTENSIFIED,
         rec-esal COLOR 5 INTENSIFIED.
ENDLOOP.

ULINE.
