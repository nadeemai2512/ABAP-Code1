REPORT Z_LAYOUT_DESIGN.

* TO CREATE FIELD VBELN OF VBAP-VBELN
DATA F_VB TYPE VBAP-VBELN.
* TO CREATE RANGE OF INPUT FIELDS
SELECT-OPTIONS S_VB FOR F_VB.
DATA : BEGIN OF IT_VBAP OCCURS 0,
 VBELN TYPE VBELN_VA,
 POSNR TYPE POSNR_VA,
 MATNR TYPE MATNR,
 NETWR TYPE NETWR_AP,
 KWMENG TYPE KWMENG,
 NETPR TYPE NETPR,
 END OF IT_VBAP.
DATA : IT_FCAT TYPE TABLE OF SLIS_FIELDCAT_ALV,
 WA_FCAT TYPE SLIS_FIELDCAT_ALV,
 ITAB_VB TYPE TABLE OF SLIS_SORTINFO_ALV,
 WA_VBAP TYPE SLIS_SORTINFO_ALV,
 WA_LAYOUT TYPE SLIS_LAYOUT_ALV.
PERFORM GET_DATA.
PERFORM BUILD_FCAT.
PERFORM SORT_ITAB.
PERFORM LAYOUT.
PERFORM DISPLAY_DATA.
FORM GET_DATA.
 SELECT VBELN POSNR MATNR NETWR KWMENG NETPR FROM VBAP
 INTO TABLE IT_VBAP WHERE VBELN IN S_VB.
 IF SY-SUBRC <> 0.
 MESSAGE 'NO RECORDS FETCHED' TYPE 'I'.
 ENDIF.
ENDFORM.
FORM BUILD_FCAT.
 DEFINE MACRO_FCAT.
 CLEAR WA_FCAT.
 WA_FCAT-COL_POS = &1.
 WA_FCAT-FIELDNAME = &2.
 WA_FCAT-TABNAME = 'IT_VBAP'.
 WA_FCAT-JUST = 'C'.
 WA_FCAT-SELTEXT_M = &3.
 IF WA_FCAT-COL_POS = 1 OR WA_FCAT-COL_POS = 2.
 WA_FCAT-KEY = 'X'.
 WA_FCAT-NO_ZERO = 'X'.
 ENDIF.
 IF WA_FCAT-COL_POS = 4 OR WA_FCAT-COL_POS = 5 OR WA_FCAT-COL_POS = 6.
 WA_FCAT-DO_SUM = 'X'.
 ENDIF.
 APPEND WA_FCAT TO IT_FCAT.
 END-OF-DEFINITION.
 MACRO_FCAT 1 'VBELN' 'SALES DOCUMENT'.
 MACRO_FCAT 2 'POSNR' 'SALES ITEM'.
 MACRO_FCAT 3 'MATNR' 'MATERIAL NUMBER'.
 MACRO_FCAT 4 'NETWR' 'NET VALUE'.
 MACRO_FCAT 5 'KWMENG' 'NET QUANTITY'.
 MACRO_FCAT 6 'NETPR' 'NET PRICE'.
ENDFORM.
FORM SORT_ITAB.
 CLEAR WA_VBAP.
 WA_VBAP-FIELDNAME = 'VBELN'.
 WA_VBAP-UP = 'X'.
 WA_VBAP-SUBTOT = 'X'.
 APPEND WA_VBAP TO ITAB_VB.
ENDFORM.
FORM LAYOUT .
 WA_LAYOUT-NO_COLHEAD = ' '.
ENDFORM.
FORM DISPLAY_DATA.
 CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
 EXPORTING
 I_CALLBACK_PROGRAM = SY-REPID
 I_GRID_TITLE = 'SALES INFORMATION'
 IS_LAYOUT = WA_LAYOUT
 IT_FIELDCAT = IT_FCAT
 IT_SORT = ITAB_VB
 TABLES
 T_OUTTAB = IT_VBAP
 EXCEPTIONS
 PROGRAM_ERROR = 1
 OTHERS = 2.
 IF SY-SUBRC <> 0.
* Implement suitable error handling here
 ENDIF.
ENDFORM.
