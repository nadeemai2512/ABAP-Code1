REPORT Z_WORKAREA.
* ========== EXAMPLE 1: BASIC WORK AREA ==========
WRITE: / '1. BASIC WORK AREA'.

DATA: BEGIN OF WA,
        ENO(3)  TYPE N,
        ENAME(14) TYPE C,
        ESAL    TYPE I,
      END OF WA.

WA-ENO = 101.
WA-ENAME = 'RAMESH'.
WA-ESAL = 9274.

WRITE: / WA-ENO, WA-ENAME, WA-ESAL.
ULINE.

* ========== EXAMPLE 2: WORK AREA WITH INITIALIZATION ==========
WRITE: / '2. WORK AREA WITH INITIALIZATION'.

DATA: BEGIN OF PRODUCT,
        PROD_ID(8)  TYPE C VALUE 'P1001',
        PROD_NAME(20) TYPE C VALUE 'LAPTOP',
        PROD_PRICE   TYPE P DECIMALS 2 VALUE 45000,
        PROD_QTY     TYPE I VALUE 10,
      END OF PRODUCT.

WRITE: / PRODUCT-PROD_ID, PRODUCT-PROD_NAME, PRODUCT-PROD_PRICE, PRODUCT-PROD_QTY.
ULINE.

* ========== EXAMPLE 3: WORK AREA WITH LIKE ==========
WRITE: / '3. WORK AREA WITH LIKE'.

DATA: BEGIN OF ADDRESS,
        CITY(20)   TYPE C,
        PIN(6)     TYPE N,
      END OF ADDRESS.

DATA: SHIP_ADDR LIKE ADDRESS.

ADDRESS-CITY = 'MUMBAI'.
ADDRESS-PIN = 400001.
SHIP_ADDR = ADDRESS.

WRITE: / SHIP_ADDR-CITY, SHIP_ADDR-PIN.
ULINE.

* ========== EXAMPLE 4: WORK AREA WITH CLEAR ==========
WRITE: / '4. WORK AREA WITH CLEAR'.

DATA: BEGIN OF STUDENT,
        ROLL_NO(4) TYPE N,
        STU_NAME(15) TYPE C,
        MARKS      TYPE I,
      END OF STUDENT.

STUDENT-ROLL_NO = 101.
STUDENT-STU_NAME = 'RAHUL'.
STUDENT-MARKS = 85.

WRITE: / 'Before Clear:', STUDENT-ROLL_NO, STUDENT-STU_NAME, STUDENT-MARKS.
CLEAR STUDENT.
WRITE: / 'After Clear:', STUDENT-ROLL_NO, STUDENT-STU_NAME, STUDENT-MARKS.
ULINE.

* ========== EXAMPLE 5: NESTED WORK AREA ==========
WRITE: / '5. NESTED WORK AREA'.

DATA: BEGIN OF COMPANY,
        COMP_NAME(20) TYPE C VALUE 'TECH SOLUTIONS',
        BEGIN OF CONTACT,
          PHONE(12)   TYPE N,
          EMAIL(25)   TYPE C,
        END OF CONTACT,
      END OF COMPANY.

COMPANY-CONTACT-PHONE = 9988776655.
COMPANY-CONTACT-EMAIL = 'info@tech.com'.

WRITE: / 'Company:', COMPANY-COMP_NAME.
WRITE: / 'Phone:', COMPANY-CONTACT-PHONE.
WRITE: / 'Email:', COMPANY-CONTACT-EMAIL.
ULINE.

SKIP.
WRITE: / 'ALL WORK AREA EXAMPLES COMPLETED'.
