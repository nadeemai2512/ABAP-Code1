REPORT Z_ALV_FIELDCATALOG25.

* TABLE DECLARATION
TABLES: MARA,
        MARC,
        MAKT,
        MARD.

* TYPE DECLARATION
TYPES: BEGIN OF TY_FINAL,

         MATNR TYPE CHAR18,
         MTART TYPE MARA-MTART,
         MATKL TYPE MARA-MATKL,
         MEINS TYPE MARA-MEINS,
         WERKS TYPE MARC-WERKS,
         LGORT TYPE MARD-LGORT,
         MAKTX TYPE MAKT-MAKTX,
         SPRAS TYPE MAKT-SPRAS,
         ERSDA TYPE MARA-ERSDA,
         ERNAM TYPE MARA-ERNAM,

       END OF TY_FINAL.

* INTERNAL TABLES
DATA: IT_FINAL TYPE TABLE OF TY_FINAL,
      WA_FINAL TYPE TY_FINAL.

* FIELD CATALOG
DATA: IT_FCAT TYPE SLIS_T_FIELDCAT_ALV,
      WA_FCAT TYPE SLIS_FIELDCAT_ALV.

* LAYOUT
DATA: WA_LAYOUT TYPE SLIS_LAYOUT_ALV.

* VARIABLE
DATA: LV_MATNR TYPE MATNR.

* SELECT-OPTIONS
SELECT-OPTIONS: S_MATNR FOR LV_MATNR.

* START-OF-SELECTION
START-OF-SELECTION.

  PERFORM FETCH_DATA.
  PERFORM BUILD_FCAT.
  PERFORM DISPLAY_ALV.

* FETCH DATA
FORM FETCH_DATA.

  SELECT A~MATNR
         A~MTART
         A~MATKL
         A~MEINS
         A~ERSDA
         A~ERNAM
         B~WERKS
         D~LGORT
         C~MAKTX
         C~SPRAS

    FROM MARA AS A

    INNER JOIN MARC AS B
    ON A~MATNR = B~MATNR

    INNER JOIN MAKT AS C
    ON A~MATNR = C~MATNR

    INNER JOIN MARD AS D
    ON A~MATNR = D~MATNR

    INTO CORRESPONDING FIELDS OF TABLE IT_FINAL

    WHERE A~MATNR IN S_MATNR
      AND C~SPRAS = SY-LANGU.

* REMOVE LEADING ZEROS
  LOOP AT IT_FINAL INTO WA_FINAL.

    SHIFT WA_FINAL-MATNR LEFT DELETING LEADING '0'.

    MODIFY IT_FINAL FROM WA_FINAL.

  ENDLOOP.

ENDFORM.
* BUILD FIELD CATALOG
FORM BUILD_FCAT.

* MATERIAL NUMBER
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 1.
  WA_FCAT-FIELDNAME = 'MATNR'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'MATERIAL NUMBER'.
  APPEND WA_FCAT TO IT_FCAT.

* MATERIAL TYPE
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 2.
  WA_FCAT-FIELDNAME = 'MTART'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'MATERIAL TYPE'.
  APPEND WA_FCAT TO IT_FCAT.

* MATERIAL GROUP
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 3.
  WA_FCAT-FIELDNAME = 'MATKL'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'MATERIAL GROUP'.
  APPEND WA_FCAT TO IT_FCAT.

* BASE UNIT
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 4.
  WA_FCAT-FIELDNAME = 'MEINS'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'BASE UNIT'.
  APPEND WA_FCAT TO IT_FCAT.

* PLANT
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 5.
  WA_FCAT-FIELDNAME = 'WERKS'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'PLANT'.
  APPEND WA_FCAT TO IT_FCAT.

* STORAGE LOCATION
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 6.
  WA_FCAT-FIELDNAME = 'LGORT'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'STORAGE LOCATION'.
  APPEND WA_FCAT TO IT_FCAT.

* DESCRIPTION
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 7.
  WA_FCAT-FIELDNAME = 'MAKTX'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-SELTEXT_M = 'DESCRIPTION'.
  APPEND WA_FCAT TO IT_FCAT.

ENDFORM.
* DISPLAY ALV
FORM DISPLAY_ALV.

  WA_LAYOUT-ZEBRA = 'X'.
  WA_LAYOUT-COLWIDTH_OPTIMIZE = 'X'.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'

    EXPORTING
      I_CALLBACK_PROGRAM = SY-REPID
      IS_LAYOUT          = WA_LAYOUT
      IT_FIELDCAT        = IT_FCAT
      I_GRID_TITLE       = 'ALV REPORT WITH FIELD CATALOG'

    TABLES
      T_OUTTAB           = IT_FINAL

    EXCEPTIONS
      PROGRAM_ERROR      = 1
      OTHERS             = 2.

  IF SY-SUBRC <> 0.
    MESSAGE 'ERROR IN ALV DISPLAY' TYPE 'E'.
  ENDIF.

ENDFORM.
