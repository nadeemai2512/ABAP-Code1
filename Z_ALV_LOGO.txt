REPORT Z_ALV_LOGO.

TYPE-POOLS: SLIS.

TABLES: MARA,
        MAKT.

*---------------------------------------------------------------------*
* INTERNAL TABLE
*---------------------------------------------------------------------*
TYPES: BEGIN OF TY_FINAL,

         MATNR TYPE MARA-MATNR,
         MTART TYPE MARA-MTART,
         MATKL TYPE MARA-MATKL,
         MEINS TYPE MARA-MEINS,
         MAKTX TYPE MAKT-MAKTX,

       END OF TY_FINAL.

DATA: IT_FINAL TYPE TABLE OF TY_FINAL,
      WA_FINAL TYPE TY_FINAL.

*---------------------------------------------------------------------*
* FIELD CATALOG
*---------------------------------------------------------------------*
DATA: IT_FCAT TYPE SLIS_T_FIELDCAT_ALV,
      WA_FCAT TYPE SLIS_FIELDCAT_ALV.

*---------------------------------------------------------------------*
* LAYOUT
*---------------------------------------------------------------------*
DATA: WA_LAYOUT TYPE SLIS_LAYOUT_ALV.

*---------------------------------------------------------------------*
* TOP OF PAGE
*---------------------------------------------------------------------*
DATA: IT_HEADER TYPE SLIS_T_LISTHEADER,
      WA_HEADER TYPE SLIS_LISTHEADER.

*---------------------------------------------------------------------*
* VARIABLES
*---------------------------------------------------------------------*
DATA: LV_MATNR TYPE MATNR.

*---------------------------------------------------------------------*
* SELECT OPTIONS
*---------------------------------------------------------------------*
SELECT-OPTIONS: S_MATNR FOR LV_MATNR.

*---------------------------------------------------------------------*
* START OF SELECTION
*---------------------------------------------------------------------*
START-OF-SELECTION.

  PERFORM FETCH_DATA.
  PERFORM BUILD_FCAT.
  PERFORM DISPLAY_ALV.

*---------------------------------------------------------------------*
* FETCH DATA
*---------------------------------------------------------------------*
FORM FETCH_DATA.

  SELECT A~MATNR
         A~MTART
         A~MATKL
         A~MEINS
         B~MAKTX

    INTO CORRESPONDING FIELDS OF TABLE IT_FINAL

    FROM MARA AS A

    INNER JOIN MAKT AS B
    ON A~MATNR = B~MATNR

    WHERE A~MATNR IN S_MATNR
      AND B~SPRAS = SY-LANGU.

ENDFORM.

*---------------------------------------------------------------------*
* BUILD FIELD CATALOG
*---------------------------------------------------------------------*
FORM BUILD_FCAT.

* MATERIAL NUMBER
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 1.
  WA_FCAT-FIELDNAME = 'MATNR'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-JUST      = 'C'.
  WA_FCAT-SELTEXT_M = 'MATERIAL NUMBER'.
  APPEND WA_FCAT TO IT_FCAT.

* MATERIAL TYPE
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 2.
  WA_FCAT-FIELDNAME = 'MTART'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-JUST      = 'C'.
  WA_FCAT-SELTEXT_M = 'MATERIAL TYPE'.
  APPEND WA_FCAT TO IT_FCAT.

* MATERIAL GROUP
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 3.
  WA_FCAT-FIELDNAME = 'MATKL'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-JUST      = 'C'.
  WA_FCAT-SELTEXT_M = 'MATERIAL GROUP'.
  APPEND WA_FCAT TO IT_FCAT.

* BASE UNIT
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 4.
  WA_FCAT-FIELDNAME = 'MEINS'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-JUST      = 'C'.
  WA_FCAT-SELTEXT_M = 'BASE UNIT'.
  APPEND WA_FCAT TO IT_FCAT.

* DESCRIPTION
  CLEAR WA_FCAT.
  WA_FCAT-COL_POS   = 5.
  WA_FCAT-FIELDNAME = 'MAKTX'.
  WA_FCAT-TABNAME   = 'IT_FINAL'.
  WA_FCAT-JUST      = 'C'.
  WA_FCAT-SELTEXT_M = 'DESCRIPTION'.
  APPEND WA_FCAT TO IT_FCAT.

ENDFORM.

*---------------------------------------------------------------------*
* DISPLAY ALV
*---------------------------------------------------------------------*
FORM DISPLAY_ALV.

  WA_LAYOUT-ZEBRA = 'X'.
  WA_LAYOUT-COLWIDTH_OPTIMIZE = 'X'.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'

    EXPORTING
      I_CALLBACK_PROGRAM = SY-REPID
      I_CALLBACK_TOP_OF_PAGE = 'TOP_OF_PAGE'
      IS_LAYOUT          = WA_LAYOUT
      IT_FIELDCAT        = IT_FCAT
      I_GRID_TITLE       = 'ALV REPORT WITH PAGE HEADING AND LOGO'

    TABLES
      T_OUTTAB           = IT_FINAL

    EXCEPTIONS
      PROGRAM_ERROR      = 1
      OTHERS             = 2.

ENDFORM.

*---------------------------------------------------------------------*
* TOP OF PAGE
*---------------------------------------------------------------------*
FORM TOP_OF_PAGE.

  CLEAR WA_HEADER.
  WA_HEADER-TYP  = 'H'.
  WA_HEADER-INFO = 'MATERIAL MASTER ALV REPORT'.
  APPEND WA_HEADER TO IT_HEADER.

  CLEAR WA_HEADER.
  WA_HEADER-TYP  = 'S'.
  WA_HEADER-KEY  = 'DATE : '.
  WA_HEADER-INFO = SY-DATUM.
  APPEND WA_HEADER TO IT_HEADER.

  CLEAR WA_HEADER.
  WA_HEADER-TYP  = 'S'.
  WA_HEADER-KEY  = 'USER : '.
  WA_HEADER-INFO = SY-UNAME.
  APPEND WA_HEADER TO IT_HEADER.

*---------------------------------------------------------------------*
* DISPLAY PAGE HEADER
*---------------------------------------------------------------------*
  CALL FUNCTION 'REUSE_ALV_COMMENTARY_WRITE'
    EXPORTING
      IT_LIST_COMMENTARY = IT_HEADER
      I_LOGO             = 'ENJOYSAP_LOGO'.

ENDFORM.
