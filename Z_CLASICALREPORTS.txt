REPORT Z_CLASICALREPORTS.

* TO CREATE A VARIABLE FOR SALES DOCUMENT FIELD OF VBAK
DATA F_SD TYPE VBELN_VA.
* TO CREATE RANGE OF INPUT FIELDS
SELECT-OPTIONS S_SD FOR F_SD.
* TO CREATE A STRUCTURE FOR FIELDS OF VBAP AND VBAK
TYPES : BEGIN OF STR,
 VBELN TYPE VBELN_VA,
 ERDAT TYPE ERDAT,
 ERNAM TYPE ERNAM,
 VBTYP TYPE VBTYP,
 TRVOG TYPE TRVOG,
 AUART TYPE AUART,
 POSNR TYPE POSNR_VA,
 MATNR TYPE MATNR,
 NETWR TYPE NETWR_AP,
 KWMENG TYPE KWMENG,
 NETPR TYPE NETPR,
 END OF STR.
DATA : SD_IT TYPE TABLE OF STR,
 WA_SD TYPE STR,
 FCAT_IT TYPE TABLE OF SLIS_FIELDCAT_ALV,
 WA_FCAT TYPE SLIS_FIELDCAT_ALV.
PERFORM FETCH_DATA.
PERFORM BUILD_FC.
PERFORM DISPLAY_DATA.
FORM FETCH_DATA.
 SELECT A~VBELN, A~ERDAT ,A~ERNAM ,A~VBTYP ,A~TRVOG ,A~AUART, B~POSNR ,B~MATNR ,B~NETWR, B~KWMENG, B~NETPR
 FROM VBAK AS A INNER JOIN VBAP AS B ON A~VBELN = B~VBELN
 INTO TABLE @SD_IT WHERE A~VBELN IN @S_SD.
 IF SY-SUBRC <> 0.
 MESSAGE 'NO RECORDS FETCHED' TYPE 'S' DISPLAY LIKE 'E'.
 ENDIF.
ENDFORM.
FORM DISPLAY_DATA.
 CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
 EXPORTING
 I_CALLBACK_PROGRAM = SY-REPID
* I_CALLBACK_TOP_OF_PAGE = 'SUBR1'
 I_GRID_TITLE = 'SALES INFORMATION'
 IT_FIELDCAT = FCAT_IT
 TABLES
 T_OUTTAB = SD_IT
 EXCEPTIONS
 PROGRAM_ERROR = 1
 OTHERS = 2.
 IF SY-SUBRC <> 0.
* Implement suitable error handling here
 ENDIF.
ENDFORM.
FORM BUILD_FC.
 DEFINE FCAT.
 CLEAR WA_FCAT.
 WA_FCAT-COL_POS = &1.
 WA_FCAT-FIELDNAME = &2.
 WA_FCAT-JUST = 'C'.
 WA_FCAT-TABNAME = 'SD_IT'.
 WA_FCAT-SELTEXT_M = &3.
 IF WA_FCAT-COL_POS = 1 OR WA_FCAT-COL_POS = 7.
 WA_FCAT-KEY = 'X'.
 WA_FCAT-NO_ZERO = 'X'.
 ENDIF.
 APPEND WA_FCAT TO FCAT_IT.
 END-OF-DEFINITION.
 FCAT 1 'VBELN' 'SALES DOCUMENT'.
 FCAT 2 'ERDAT' 'DATE OF CREATION'.
 FCAT 3 'ERNAM' 'PERSON CREATED'.
 FCAT 4 'VBTYP' 'SD DOC CATEGORY'.
 FCAT 5 'TRVOG' 'TRANSACTION GRP'.
 FCAT 6 'AUART' 'SALES DOC TYPE'.
 FCAT 7 'POSNR' 'SALES DOC ITEM'.
 FCAT 8 'MATNR' 'MATERIAL NUM'.
 FCAT 9 'NETWR' 'NET VALUE OF ORDER ITEM'.
 FCAT 10 'KWMENG' 'CUMM QTY IN SALES UNIT'.
 FCAT 11 'NETPR' 'NET PRICE'.
ENDFORM.
