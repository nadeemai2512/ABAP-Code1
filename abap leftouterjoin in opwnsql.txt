REPORT Z_FILLING_FINAL.

TABLES: MARA, MARC.

* TYPE DECLARATION
TYPES: BEGIN OF TY_MARA,
         MATNR TYPE MARA-MATNR,
         MTART TYPE MARA-MTART,
         MEINS TYPE MARA-MEINS,
       END OF TY_MARA.

TYPES: BEGIN OF TY_MARC,
         MATNR TYPE MARC-MATNR,
         WERKS TYPE MARC-WERKS,
       END OF TY_MARC.

TYPES: BEGIN OF TY_FINAL,
         MATNR TYPE MARA-MATNR,
         MTART TYPE MARA-MTART,
         MEINS TYPE MARA-MEINS,
         WERKS TYPE MARC-WERKS,
       END OF TY_FINAL.

* INTERNAL TABLES & WORK AREAS
DATA: IT_MARA  TYPE TABLE OF TY_MARA,
      WA_MARA  TYPE TY_MARA,

      IT_MARC  TYPE TABLE OF TY_MARC,
      WA_MARC  TYPE TY_MARC,

      IT_FINAL TYPE TABLE OF TY_FINAL,
      WA_FINAL TYPE TY_FINAL.

* HEADING
FORMAT COLOR 5 INTENSIFIED ON.
WRITE:/ 'FILL FINAL INTERNAL TABLE FROM IT_MARA AND IT_MARC'.
ULINE.

* SELECT DATA FROM MARA
SELECT MATNR
       MTART
       MEINS
INTO TABLE IT_MARA
FROM MARA
WHERE MATNR BETWEEN '000000000000000001'
                AND '000000000000001000'.

* SELECT DATA FROM MARC
IF IT_MARA IS NOT INITIAL.

  SELECT MATNR
         WERKS
  INTO TABLE IT_MARC
  FROM MARC
  FOR ALL ENTRIES IN IT_MARA
  WHERE MATNR = IT_MARA-MATNR.

ENDIF.

* FILL FINAL INTERNAL TABLE
LOOP AT IT_MARA INTO WA_MARA.

  LOOP AT IT_MARC INTO WA_MARC
       WHERE MATNR = WA_MARA-MATNR.

    CLEAR WA_FINAL.

    WA_FINAL-MATNR = WA_MARA-MATNR.
    WA_FINAL-MTART = WA_MARA-MTART.
    WA_FINAL-MEINS = WA_MARA-MEINS.
    WA_FINAL-WERKS = WA_MARC-WERKS.

    APPEND WA_FINAL TO IT_FINAL.

  ENDLOOP.

ENDLOOP.

* DISPLAY NUMBER OF RECORDS
FORMAT COLOR 3 INTENSIFIED ON.
WRITE:/ 'NO. OF RECORDS IN FINAL TABLE = ', LINES( IT_FINAL ).
ULINE.

* COLUMN HEADINGS
FORMAT COLOR 6 INTENSIFIED ON.

WRITE:/
'MATNR', 20
'MTART', 35
'MEINS', 50
'WERKS'.

ULINE.

* DISPLAY OUTPUT
FORMAT COLOR 2 INTENSIFIED OFF.

LOOP AT IT_FINAL INTO WA_FINAL.

  WRITE:/
  WA_FINAL-MATNR, 20
  WA_FINAL-MTART, 35
  WA_FINAL-MEINS, 50
  WA_FINAL-WERKS.

ENDLOOP.

FORMAT RESET.
