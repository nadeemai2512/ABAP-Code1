*&      Form  FETCH_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM FETCH_DATA .

  SELECT MATNR ERSDA ERNAM MTART MBRSH MATKL MEINS FROM MARA INTO TABLE IT_MARA
    WHERE MTART IN S_MTART.

  IF IT_MARA IS NOT INITIAL.
    SELECT MATNR WERKS EKGRP STEUC FROM MARC INTO TABLE IT_MARC
      FOR ALL ENTRIES IN IT_MARA WHERE MATNR = IT_MARA-MATNR AND WERKS IN S_WERKS
      AND STEUC IN S_STEUC.

    SELECT MATNR SPRAS MAKTX FROM MAKT INTO TABLE IT_MAKT FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR AND SPRAS = SY-LANGU.
  ENDIF.

  IF IT_MARA IS NOT INITIAL.
    SELECT MATNR BWKEY BWTAR BKLAS FROM MBEW INTO TABLE IT_MBEW FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR.
  ENDIF.

  IF IT_MARC IS NOT INITIAL.
    SELECT MATNR VKORG VTWEG DWERK KTGRM FROM MVKE INTO TABLE IT_MVKE FOR ALL ENTRIES IN IT_MARC
      WHERE MATNR = IT_MARC-MATNR AND DWERK = IT_MARC-WERKS.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  SORTING
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM SORTING .
*  SORT ALL INTERNAL TABLE WITH KEY FIELDS FOR BINARY SEARCH
  SORT : IT_MARA BY MATNR,
         IT_MARC BY MATNR WERKS,
         IT_MVKE BY MATNR VKORG VTWEG,
         IT_MAKT BY MATNR,
         IT_MBEW BY MATNR BWKEY.
ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  PROCESS_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM PROCESS_DATA .
  LOOP AT IT_MARC INTO WA_MARC.

    WA_FINAL-MATNR = WA_MARC-MATNR.
    WA_FINAL-WERKS = WA_MARC-WERKS.
    WA_FINAL-EKGRP = WA_MARC-EKGRP.
    WA_FINAL-STEUC = WA_MARC-STEUC.

    READ TABLE IT_MARA INTO WA_MARA WITH KEY MATNR = WA_MARC-MATNR BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-ERSDA = WA_MARA-ERSDA.
      WA_FINAL-ERNAM = WA_MARA-ERNAM.
      WA_FINAL-MTART = WA_MARA-MTART.
      WA_FINAL-MBRSH = WA_MARA-MBRSH.
      WA_FINAL-MATKL = WA_MARA-MATKL.
      WA_FINAL-MEINS = WA_MARA-MEINS.
    ENDIF.

    READ TABLE IT_MAKT INTO WA_MAKT WITH KEY MATNR = WA_MARC-MATNR SPRAS = SY-LANGU BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-SPRAS = WA_MAKT-SPRAS.
      WA_FINAL-MAKTX = WA_MAKT-MAKTX.
    ENDIF.

    READ TABLE IT_MBEW INTO WA_MBEW WITH KEY MATNR = WA_MARC-MATNR BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-BWKEY = WA_MBEW-BWKEY.
*      WA_FINAL-BWTAR = WA_MBEW-BWTAR.
      WA_FINAL-BKLAS = WA_MBEW-BKLAS.
    ENDIF.

    LOOP AT IT_MVKE INTO WA_MVKE WHERE MATNR = WA_MARC-MATNR AND DWERK = WA_MARC-WERKS.
      WA_FINAL-VKORG = WA_MVKE-VKORG.
      WA_FINAL-VTWEG = WA_MVKE-VTWEG.
      WA_FINAL-DWERK = WA_MVKE-DWERK.
      WA_FINAL-KTGRM = WA_MVKE-KTGRM.
    ENDLOOP.
    APPEND WA_FINAL TO IT_FINAL.
    CLEAR :WA_FINAL,WA_MARA,WA_MARC,WA_MAKT,WA_MBEW,WA_MVKE.
  ENDLOOP.
ENDFORM.

FORM BUILD_FCAT .

  DEFINE MACRO_FCAT.
    CLEAR WA_FCAT.

    WA_FCAT-COL_POS = &1.
    WA_FCAT-FIELDNAME = &2.
    WA_FCAT-JUST = 'C'.
    WA_FCAT-TABNAME = 'IT_FINAL'.
    WA_FCAT-SELTEXT_M = &3.

    IF WA_FCAT-COL_POS = 1 OR WA_FCAT-COL_POS = 3 OR WA_FCAT-COL_POS = 4 OR WA_FCAT-COL_POS = 10 OR WA_FCAT-COL_POS = 18.
      WA_FCAT-KEY = 'X'.
      ENDIF.

      IF WA_FCAT-COL_POS = 2.
        WA_FCAT-EMPHASIZE = 'C410'.
        ENDIF.

    APPEND WA_FCAT TO IT_FCAT.
  END-OF-DEFINITION.

  MACRO_FCAT 1 'MATNR' 'MATERIAL NUMBER'.

  MACRO_FCAT 9 'ERNAM' 'CREATED BY'.
  MACRO_FCAT 10 'WERKS' 'PLANT'.
  MACRO_FCAT 11 'MBRSH' 'INDUSTRY SECTOR'.
  MACRO_FCAT 12 'MATKL' 'MATERIAL GROUP'.
