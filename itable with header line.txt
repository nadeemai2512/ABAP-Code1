*---------------------------------------------------------------------*
* TYPE DECLARATION
*---------------------------------------------------------------------*
TYPES: BEGIN OF ty_employee,
         emp_id    TYPE numc3,

*---------------------------------------------------------------------*
* 2. INSERT OPERATION - Inserting records at specific positions
*---------------------------------------------------------------------*
CLEAR gt_employee.
gt_employee-emp_id = 108.
gt_employee-emp_name = 'Deepak Joshi'.
gt_employee-emp_city = 'Jaipur'.
gt_employee-emp_sal = 41000.
gt_employee-emp_dept = 'Sales'.
INSERT gt_employee INTO gt_employee INDEX 3.

gt_employee-emp_id = 109.
gt_employee-emp_name = 'Kavita Nair'.
gt_employee-emp_city = 'Ahmedabad'.
gt_employee-emp_sal = 46000.
gt_employee-emp_dept = 'IT'.
INSERT gt_employee INTO gt_employee INDEX 6.

FORMAT COLOR COL_HEADING.
WRITE: / '========== 2. AFTER INSERT OPERATION (Added at Index 3 & 6) =========='.
FORMAT COLOR OFF.
FORMAT COLOR COL_GROUP.
WRITE: / 'INDEX', 'ID', 'NAME', 'SALARY', 'DEPT'.
FORMAT COLOR OFF.
LOOP AT gt_employee.
  IF gt_employee-emp_id = 108 OR gt_employee-emp_id = 109.
    FORMAT COLOR COL_POSITIVE.
  ELSEIF sy-tabix MOD 2 = 0.
    FORMAT COLOR COL_KEY.
  ELSE.
    FORMAT COLOR COL_NORMAL.
  ENDIF.
  WRITE: / sy-tabix, gt_employee-emp_id, gt_employee-emp_name,
           gt_employee-emp_sal, gt_employee-emp_dept.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 3. COLLECT OPERATION - Summing salaries by department
*---------------------------------------------------------------------*
DATA: lt_dept_sum TYPE TABLE OF ty_employee WITH HEADER LINE.
REFRESH lt_dept_sum.

LOOP AT gt_employee.
  lt_dept_sum-emp_dept = gt_employee-emp_dept.
  lt_dept_sum-emp_sal = gt_employee-emp_sal.
  COLLECT lt_dept_sum.
ENDLOOP.

FORMAT COLOR COL_HEADING.
WRITE: / '========== 3. COLLECT OPERATION (Dept-wise Salary Sum) =========='.
FORMAT COLOR OFF.
FORMAT COLOR COL_GROUP.
WRITE: / 'Department', 'Total Salary'.
FORMAT COLOR OFF.
LOOP AT lt_dept_sum.
  IF lt_dept_sum-emp_sal > 100000.
    FORMAT COLOR COL_POSITIVE.
  ELSEIF lt_dept_sum-emp_sal > 50000.
    FORMAT COLOR COL_KEY.
  ELSE.
    FORMAT COLOR COL_NORMAL.
  ENDIF.
  WRITE: / lt_dept_sum-emp_dept,
          lt_dept_sum-emp_sal.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 4. APPEND LINES OPERATION - Copying range of records to backup table
*---------------------------------------------------------------------*
REFRESH gt_backup.
APPEND LINES OF gt_employee FROM 2 TO 5 TO gt_backup.

FORMAT COLOR COL_HEADING.
WRITE: / '========== 4. APPEND LINES (Records 2 to 5 to Backup Table) =========='.
FORMAT COLOR OFF.
FORMAT COLOR COL_GROUP.
WRITE: / 'INDEX', 'ID', 'NAME', 'CITY', 'SALARY'.
FORMAT COLOR OFF.
LOOP AT gt_backup.
  IF sy-tabix MOD 2 = 0.
    FORMAT COLOR COL_KEY.
  ELSE.
    FORMAT COLOR COL_NORMAL.
  ENDIF.
  WRITE: / sy-tabix, gt_backup-emp_id, gt_backup-emp_name,
           gt_backup-emp_city, gt_backup-emp_sal.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 6. READ TABLE OPERATION - Reading records (multiple methods)
*---------------------------------------------------------------------*
FORMAT COLOR COL_HEADING.
WRITE: / '========== 6. READ TABLE OPERATION =========='.
FORMAT COLOR OFF.

* Method 1: Reading by index
CLEAR gt_employee.
READ TABLE gt_employee INDEX 2.
FORMAT COLOR COL_GROUP.
WRITE: / 'Method 1 - Reading by Index 2:'.
FORMAT COLOR COL_KEY.
WRITE: / 'ID:', gt_employee-emp_id,
        'Name:', gt_employee-emp_name,
        'Salary:', gt_employee-emp_sal.

* Method 2: Reading with key (using binary search after sorting)
SORT gt_employee BY emp_id.
CLEAR gt_employee.
READ TABLE gt_employee WITH KEY emp_id = 105 BINARY SEARCH.
IF sy-subrc = 0.
  FORMAT COLOR COL_GROUP.
  WRITE: / 'Method 2 - Reading with Binary Search (ID 105):'.
  FORMAT COLOR COL_POSITIVE.
  WRITE: / 'Record found at index:', sy-tabix.
  WRITE: / 'Name:', gt_employee-emp_name, 'Dept:', gt_employee-emp_dept.
ELSE.
  FORMAT COLOR COL_NEGATIVE.
  WRITE: / 'Record with ID 105 not found'.
ENDIF.

* Method 3: Reading with non-unique key
CLEAR gt_employee.
READ TABLE gt_employee WITH KEY emp_dept = 'IT'.
FORMAT COLOR COL_GROUP.
WRITE: / 'Method 3 - Reading with Non-Unique Key (First IT Dept):'.
FORMAT COLOR COL_KEY.
WRITE: / 'ID:', gt_employee-emp_id, 'Name:', gt_employee-emp_name.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 7. LOOP OPERATION - Various loop conditions
*---------------------------------------------------------------------*
FORMAT COLOR COL_HEADING.

FORMAT COLOR OFF.

FORMAT COLOR COL_GROUP.
WRITE: / 'IT Department Employees:'.
FORMAT COLOR COL_KEY.
LOOP AT gt_employee WHERE emp_dept = 'IT'.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_sal.
ENDLOOP.

FORMAT COLOR COL_GROUP.
WRITE: / 'Employees with Salary > 45000:'.
FORMAT COLOR COL_POSITIVE.
LOOP AT gt_employee WHERE emp_sal > 45000.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_sal.
ENDLOOP.

FORMAT COLOR COL_GROUP.
WRITE: / 'Employees from Mumbai or Delhi:'.
FORMAT COLOR COL_KEY.
LOOP AT gt_employee WHERE emp_city = 'Mumbai' OR emp_city = 'Delhi'.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_city.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 14. ADDITIONAL COLLECT EXAMPLE - City wise salary sum
*---------------------------------------------------------------------*
DATA: lt_city_sum TYPE TABLE OF ty_employee WITH HEADER LINE.
REFRESH lt_city_sum.

LOOP AT gt_copy.
  lt_city_sum-emp_city = gt_copy-emp_city.
  lt_city_sum-emp_sal = gt_copy-emp_sal.
  COLLECT lt_city_sum.
ENDLOOP.

FORMAT COLOR COL_HEADING.
WRITE: / '========== 14. ADDITIONAL COLLECT (City-wise Salary Sum) =========='.
FORMAT COLOR COL_GROUP.
WRITE: / 'City', 'Total Salary'.
LOOP AT lt_city_sum.
  IF lt_city_sum-emp_sal > 100000.
    FORMAT COLOR COL_POSITIVE.
  ELSEIF lt_city_sum-emp_sal > 50000.
    FORMAT COLOR COL_KEY.
  ELSE.
    FORMAT COLOR COL_NORMAL.
  ENDIF.
  WRITE: / lt_city_sum-emp_city, lt_city_sum-emp_sal.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 15. COMPLEX LOOP WITH CALCULATIONS
*---------------------------------------------------------------------*
lv_total_sal = 0.
lv_count = 0.

LOOP AT gt_copy.
  lv_total_sal = lv_total_sal + gt_copy-emp_sal.
  lv_count = lv_count + 1.
ENDLOOP.

FORMAT COLOR COL_HEADING.
WRITE: / '========== 15. LOOP WITH CALCULATIONS =========='.
FORMAT COLOR COL_GROUP.
WRITE: / 'Total Employees:', lv_count.
IF lv_count > 0.
  WRITE: / 'Total Salary:', lv_total_sal.
ENDIF.
FORMAT COLOR OFF.
ULINE.

*---------------------------------------------------------------------*
* 16. SORT AND DISPLAY
*---------------------------------------------------------------------*
SORT gt_copy BY emp_sal DESCENDING.
FORMAT COLOR COL_HEADING.
WRITE: / '========== 16. SORTED TABLE (By Salary Descending) =========='.
FORMAT COLOR COL_GROUP.
WRITE: / 'Top 3 Highest Paid Employees:'.
FORMAT COLOR COL_POSITIVE.
lv_count = 0.
LOOP AT gt_copy.
  lv_count = lv_count + 1.
  WRITE: / gt_copy-emp_id, gt_copy-emp_name, gt_copy-emp_sal.
  IF lv_count >= 3.
    EXIT.
  ENDIF.
ENDLOOP.
FORMAT COLOR OFF.
ULINE.
