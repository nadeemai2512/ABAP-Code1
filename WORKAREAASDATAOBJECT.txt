REPORT Z_WORKAREAASDATAOBJECT.

*---------------------------------------------------------------------*
* EXAMPLE 1: STUDENT
*---------------------------------------------------------------------*
WRITE: / 'WORK AREA AS DATA OBJECT - STUDENT'.
ULINE.

DATA: BEGIN OF wa1,
        sid   TYPE n LENGTH 3,
        sname TYPE c LENGTH 15,
        sfee  TYPE i,
        sdob  TYPE d,
        scity TYPE c LENGTH 13,
      END OF wa1.

wa1-sid   = 111.
wa1-sname = 'Naruto'.
wa1-sfee  = 23493.
wa1-sdob  = '20040522'.
wa1-scity = 'SIDDIPET'.

WRITE: / 'STUDENT ID   :', wa1-sid,
       / 'STUDENT NAME :', wa1-sname,
       / 'STUDENT FEE  :', wa1-sfee,
       / 'STUDENT DOB  :', wa1-sdob,
       / 'STUDENT CITY :', wa1-scity.

ULINE.
