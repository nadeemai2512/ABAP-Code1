REPORT ZBDC_VA02_CUSTPO_UPDATE.

types:BEGIN OF line_type,
  VBELN TYPE VBELN_VA,
  BSTKD TYPE BSTKD,
  BSTDK(10) ,
  ZTERM TYPE DZTERM,
  END OF LINE_TYPE.
  DATA:M(2),D(2),Y(4),DATE1 LIKE SY-DATUM,F_DATE(10).

*  CREATING WORK AREA.
  DATA:WA TYPE LINE_TYPE,
        WA_BDCDATA TYPE BDCDATA.

*  CREATING INTERNAL TABLE.
  DATA:ITAB TYPE TABLE OF LINE_TYPE,
       IT_RAW TYPE  TRUXS_T_TEXT_DATA,
       IT_BDCDATA TYPE TABLE OF BDCDATA.

  PARAMETERS S_FILE TYPE RLGRAP-FILENAME.

AT SELECTION-SCREEN ON VALUE-REQUEST FOR S_FILE.
  PERFORM F4_FUNCTION.

START-OF-SELECTION.
PERFORM UPLOAD_FLATFILE.
PERFORM MAP_DATA.
FORM UPLOAD_FLATFILE .
  CALL FUNCTION 'TEXT_CONVERT_XLS_TO_SAP'
    EXPORTING
      I_TAB_RAW_DATA             = IT_RAW
      I_FILENAME                 = S_FILE
    TABLES
      I_TAB_CONVERTED_DATA       = ITAB.
*   EXCEPTIONS
*     CONVERSION_FAILED          = 1
*     OTHERS                     = 2
            .
  IF SY-SUBRC = 0.
    LOOP AT  ITAB INTO WA.
      WRITE : / WA-VBELN.
    ENDLOOP.

  ENDIF.


ENDFORM.

FORM F4_FUNCTION .
  CALL FUNCTION 'F4_FILENAME'
*   EXPORTING
*     PROGRAM_NAME        = SYST-CPROG
*     DYNPRO_NUMBER       = SYST-DYNNR
*     FIELD_NAME          = ' '
   IMPORTING
     FILE_NAME           = S_FILE
     .
ENDFORM.

FORM MAP_DATA .
  LOOP AT ITAB INTO WA.

Y = WA-BSTDK+6(4).
M = WA-BSTDK+3(2).
D = WA-BSTDK+0(2).
*M = WA-BSTDK+3(2).
*Y = WA-BSTDK+6(4).

CONCATENATE Y M D INTO DATE1.



  perform bdc_dynpro      using 'SAPMV45A' '0102'.
perform bdc_field       using 'BDC_CURSOR'
                              'VBAK-VBELN'.
perform bdc_field       using 'BDC_OKCODE'
                              '=SUCH'.
perform bdc_field       using 'VBAK-VBELN'
                             WA-VBELN." '0000004971'.
perform bdc_dynpro      using 'SAPMV45A' '4001'.
perform bdc_field       using 'BDC_OKCODE'
                              '=SICH'.
perform bdc_field       using 'VBKD-BSTKD'
                              WA-BSTKD."'PSIT_123'.
WRITE DATE1 to F_DATE.
perform bdc_field       using 'VBKD-BSTDK'
                              F_DATE."'06.07.2026'.
perform bdc_field       using 'KUWEV-KUNNR'
                              '1001'.
perform bdc_field       using 'BDC_CURSOR'
                              'VBKD-ZTERM'.
perform bdc_field       using 'RV45A-KETDAT'
                              '09.01.1997'.
perform bdc_field       using 'RV45A-KPRGBZ'
                              'D'.
perform bdc_field       using 'VBKD-PRSDT'
                              '07.01.1997'.
perform bdc_field       using 'VBKD-ZTERM'
                              WA-ZTERM."'ZB02'.
perform bdc_field       using 'VBKD-INCO1'
                              'CIP'.
perform bdc_field       using 'VBKD-INCO2'
                              'Frankfurt'.


 ENDLOOP.


ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  BDC_DYNPRO
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_0109   text
*      -->P_0110   text
*----------------------------------------------------------------------*
FORM BDC_DYNPRO  USING  PROG SCRNO.

  CLEAR WA_BDCDATA.
  WA_BDCDATA-PROGRAM = PROG.
  WA_BDCDATA-DYNPRO = SCRNO.
  WA_BDCDATA-DYNBEGIN = 'X'.
  APPEND WA_BDCDATA TO IT_BDCDATA.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  BDC_FIELD
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_0124   text
*      -->P_WA_VBELN  text
*----------------------------------------------------------------------*
FORM BDC_FIELD  USING   FNAME FVALUE.
  CLEAR WA_BDCDATA.
  WA_BDCDATA-FNAM = FNAME.
  WA_BDCDATA-FVAL = FVALUE.
  APPEND WA_BDCDATA TO IT_BDCDATA.

ENDFORM.
