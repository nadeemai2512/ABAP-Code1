FORM FETCH_DATA .
 SELECT A~MATNR A~ERSDA A~ERNAM A~MTART B~WERKS B~EKGRP B~MINBE B~BSTMI C~LGORT C~LABST
 FROM MARA AS A INNER JOIN MARC AS B ON A~MATNR = B~MATNR
 INNER JOIN MARD AS C ON A~MATNR = C~MATNR AND B~WERKS = C~WERKS
 INTO TABLE MAT_IT
 WHERE A~MATNR IN S_MATNR.
 IF SY-SUBRC <> 0.
 MESSAGE 'INVALID MATERIAL RANGE' TYPE 'E'.
 ENDIF.
ENDFORM.
