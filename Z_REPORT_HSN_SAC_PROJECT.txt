*&---------------------------------------------------------------------*
*& FORM : FORM_VALIDATIONS
*& Validate selection screen inputs against SAP configuration tables
*&---------------------------------------------------------------------*
FORM FORM_VALIDATIONS.

  " --- Validate Material Type against T134 ---
  IF S_MTART IS NOT INITIAL.
    SELECT SINGLE MTART
      FROM T134
      INTO WA_VAL_MTART
      WHERE MTART IN S_MTART.
    IF SY-SUBRC <> 0.
      MESSAGE 'Invalid Material Type entered. Please check and re-enter.'
              TYPE 'E'.
    ENDIF.
  ENDIF.

  " --- Validate Control Code (HSN/SAC) against T604F ---
  IF S_STEUC IS NOT INITIAL.
    SELECT SINGLE STEUC
      FROM T604F
      INTO WA_VAL_STEUC
      WHERE STEUC IN S_STEUC.
    IF SY-SUBRC <> 0.
      MESSAGE 'Invalid Control Code (HSN/SAC) entered. Please check and re-enter.'
              TYPE 'E'.
    ENDIF.
  ENDIF.

  " --- Validate Plant against T001W ---
  IF S_WERKS IS NOT INITIAL.
    SELECT SINGLE WERKS
      FROM T001W
      INTO WA_VAL_WERKS
      WHERE WERKS IN S_WERKS.
    IF SY-SUBRC <> 0.
      MESSAGE 'Invalid Plant entered. Please check and re-enter.'
              TYPE 'E'.
    ENDIF.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& FORM : FORM_FETCH_DATA
*& Fetch data from DB tables in sequence:
*&   MARA -> MARC -> MAKT, MBEW, MVKE
*&---------------------------------------------------------------------*
FORM FORM_FETCH_DATA.

  " ---------------------------------------------------------------
  " STEP 1: Fetch General Material Data from MARA
  "         Filter by Material Type (S_MTART)
  " ---------------------------------------------------------------
  SELECT MATNR ERSDA ERNAM MTART MBRSH MATKL MEINS
    FROM MARA
    INTO TABLE IT_MARA
    WHERE MTART IN S_MTART.

  IF IT_MARA IS INITIAL.
    MESSAGE 'No materials found for the selected Material Type. Execution stopped.'
            TYPE 'S' DISPLAY LIKE 'W'.
    STOP.
  ENDIF.

  " ---------------------------------------------------------------
  " STEP 2: Fetch Plant Data from MARC
  "         Filter by Plant (S_WERKS) and Control Code (S_STEUC)
  "         FOR ALL ENTRIES based on MARA results
  " ---------------------------------------------------------------
  IF IT_MARA IS NOT INITIAL.
    SELECT MATNR WERKS EKGRP STEUC
      FROM MARC
      INTO TABLE IT_MARC
      FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR
        AND WERKS IN S_WERKS
        AND STEUC IN S_STEUC.
  ENDIF.

  IF IT_MARC IS INITIAL.
    MESSAGE 'No plant or control code data found for the given selection.'
            TYPE 'S' DISPLAY LIKE 'W'.
    STOP.
  ENDIF.

  " ---------------------------------------------------------------
  " STEP 3: Fetch Material Descriptions from MAKT
  "         Language = current logon language (SY-LANGU)
  "         FOR ALL ENTRIES based on MARC results
  " ---------------------------------------------------------------
  IF IT_MARC IS NOT INITIAL.
    SELECT MATNR MAKTX SPRAS
      FROM MAKT
      INTO TABLE IT_MAKT
      FOR ALL ENTRIES IN IT_MARC
      WHERE MATNR = IT_MARC-MATNR
        AND SPRAS = SY-LANGU.
  ENDIF.

  " ---------------------------------------------------------------
  " STEP 4: Fetch Material Valuation from MBEW
  "         BWKEY (Valuation Area) = WERKS (Plant)
  "         FOR ALL ENTRIES based on MARC results
  " ---------------------------------------------------------------
  IF IT_MARC IS NOT INITIAL.
    SELECT MATNR BWKEY BWTAR BKLAS
      FROM MBEW
      INTO TABLE IT_MBEW
      FOR ALL ENTRIES IN IT_MARC
      WHERE MATNR = IT_MARC-MATNR
        AND BWKEY = IT_MARC-WERKS.
  ENDIF.

  " ---------------------------------------------------------------
  " STEP 5: Fetch Sales Data from MVKE
  "         DWERK (Delivering Plant) = WERKS (Plant from MARC)
  "         FOR ALL ENTRIES based on MARC results
  " ---------------------------------------------------------------
  IF IT_MARC IS NOT INITIAL.
    SELECT MATNR VKORG VTWEG DWERK KTGRM
      FROM MVKE
      INTO TABLE IT_MVKE
      FOR ALL ENTRIES IN IT_MARC
      WHERE MATNR = IT_MARC-MATNR
        AND DWERK = IT_MARC-WERKS.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& FORM : FORM_PROCESS_DATA
*& Merge all internal tables into IT_FINAL
*& Primary loop: IT_MARC (drives plant + control code combination)
*& All READ TABLE use BINARY SEARCH for performance
*&---------------------------------------------------------------------*
FORM FORM_PROCESS_DATA.

  " Sort all tables by their key fields BEFORE binary search
  SORT IT_MARA BY MATNR.
  SORT IT_MARC BY MATNR WERKS.
  SORT IT_MAKT BY MATNR.
  SORT IT_MBEW BY MATNR BWKEY.
  SORT IT_MVKE BY MATNR DWERK.

  LOOP AT IT_MARC INTO WA_MARC.

    CLEAR WA_FINAL.  " Clear work area at start of every iteration

    " --- Populate MARC fields ---
    WA_FINAL-MATNR = WA_MARC-MATNR.
    WA_FINAL-WERKS = WA_MARC-WERKS.
    WA_FINAL-EKGRP = WA_MARC-EKGRP.
    WA_FINAL-STEUC = WA_MARC-STEUC.

    " --- Read MARA: General Material Data ---
    READ TABLE IT_MARA INTO WA_MARA
      WITH KEY MATNR = WA_MARC-MATNR
      BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-MTART = WA_MARA-MTART.   " Material Type
      WA_FINAL-MBRSH = WA_MARA-MBRSH.   " Industry Sector
      WA_FINAL-MATKL = WA_MARA-MATKL.   " Material Group
      WA_FINAL-MEINS = WA_MARA-MEINS.   " Unit of Measure
      WA_FINAL-ERSDA = WA_MARA-ERSDA.   " Created On
      WA_FINAL-ERNAM = WA_MARA-ERNAM.   " Created By
    ENDIF.

    " --- Read MAKT: Material Description (language-dependent) ---
    READ TABLE IT_MAKT INTO WA_MAKT
      WITH KEY MATNR = WA_MARC-MATNR
      BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-MAKTX = WA_MAKT-MAKTX.   " Material Description
    ENDIF.

    " --- Read MBEW: Valuation Data (key: MATNR + BWKEY=WERKS) ---
    READ TABLE IT_MBEW INTO WA_MBEW
      WITH KEY MATNR = WA_MARC-MATNR
               BWKEY = WA_MARC-WERKS
      BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-BWKEY = WA_MBEW-BWKEY.   " Valuation Area
      WA_FINAL-BWTAR = WA_MBEW-BWTAR.   " Valuation Type
      WA_FINAL-BKLAS = WA_MBEW-BKLAS.   " Valuation Class
    ENDIF.

    " --- Read MVKE: Sales Data (key: MATNR + DWERK=WERKS) ---
    READ TABLE IT_MVKE INTO WA_MVKE
      WITH KEY MATNR = WA_MARC-MATNR
               DWERK = WA_MARC-WERKS
      BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-VKORG = WA_MVKE-VKORG.   " Sales Organization
      WA_FINAL-VTWEG = WA_MVKE-VTWEG.   " Distribution Channel
      WA_FINAL-DWERK = WA_MVKE-DWERK.   " Delivering Plant
      WA_FINAL-KTGRM = WA_MVKE-KTGRM.   " Account Assignment Group
    ENDIF.

    APPEND WA_FINAL TO IT_FINAL.

  ENDLOOP.

  IF IT_FINAL IS INITIAL.
    MESSAGE 'No output data to display after processing.' TYPE 'S' DISPLAY LIKE 'W'.
    STOP.
  ENDIF.

ENDFORM.

    EXCEPTIONS

    MESSAGE 'Error occurred while displaying the ALV Grid report.' TYPE 'E'.
  ENDIF.
