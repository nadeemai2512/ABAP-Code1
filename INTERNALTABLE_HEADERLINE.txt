           gt_employee-emp_sal, gt_employee-emp_dept.

ULINE.

* 3. COLLECT OPERATION - Summing salaries by department
DATA: lt_dept_sum TYPE TABLE OF ty_employee WITH HEADER LINE.
REFRESH lt_dept_sum.

LOOP AT gt_employee.
  lt_dept_sum-emp_dept = gt_employee-emp_dept.
  lt_dept_sum-emp_sal = gt_employee-emp_sal.
  COLLECT lt_dept_sum.
ENDLOOP.

WRITE: / '========== 3. COLLECT OPERATION (Dept-wise Salary Sum) =========='.
LOOP AT lt_dept_sum.
  WRITE: / 'Department:', lt_dept_sum-emp_dept,
          'Total Salary:', lt_dept_sum-emp_sal.
ENDLOOP.
ULINE.

* 4. APPEND LINES OPERATION - Copying range of records to backup table
REFRESH gt_backup.
APPEND LINES OF gt_employee FROM 2 TO 5 TO gt_backup.

WRITE: / '========== 4. APPEND LINES (Records 2 to 5 to Backup Table) =========='.
LOOP AT gt_backup.
  WRITE: / sy-tabix, gt_backup-emp_id, gt_backup-emp_name,
           gt_backup-emp_city, gt_backup-emp_sal.
ENDLOOP.
ULINE.

* 6. READ TABLE OPERATION - Reading records (multiple methods)
* Method 1: Reading by index
CLEAR gt_employee.
READ TABLE gt_employee INDEX 2.
WRITE: / '========== 6. READ TABLE OPERATION =========='.
WRITE: / 'Record at Index 2:'.
WRITE: / 'ID:', gt_employee-emp_id,
        'Name:', gt_employee-emp_name,
        'Salary:', gt_employee-emp_sal.

* Method 2: Reading with key (using binary search after sorting)
SORT gt_employee BY emp_id.
CLEAR gt_employee.
READ TABLE gt_employee WITH KEY emp_id = 105 BINARY SEARCH.
IF sy-subrc = 0.
  WRITE: / 'Record with ID 105 found at index:', sy-tabix.
  WRITE: / 'Name:', gt_employee-emp_name, 'Dept:', gt_employee-emp_dept.
ELSE.
  WRITE: / 'Record with ID 105 not found'.
ENDIF.

* Method 3: Reading with non-unique key
CLEAR gt_employee.
READ TABLE gt_employee WITH KEY emp_dept = 'IT'.
WRITE: / 'First IT department employee found:'.
WRITE: / 'ID:', gt_employee-emp_id, 'Name:', gt_employee-emp_name.
ULINE.

* 7. LOOP OPERATION - Various loop conditions
WRITE: / '========== 7. LOOP OPERATIONS =========='.

WRITE: / 'IT Department Employees:'.
LOOP AT gt_employee WHERE emp_dept = 'IT'.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_sal.
ENDLOOP.

WRITE: / 'Employees with Salary > 45000:'.
LOOP AT gt_employee WHERE emp_sal > 45000.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_sal.
ENDLOOP.

WRITE: / 'Employees from Mumbai or Delhi:'.
LOOP AT gt_employee WHERE emp_city = 'Mumbai' OR emp_city = 'Delhi'.
  WRITE: / gt_employee-emp_id, gt_employee-emp_name, gt_employee-emp_city.
ENDLOOP.
ULINE.

* 9. MOVE OPERATION - Copying entire table
DATA: gt_copy TYPE TABLE OF ty_employee WITH HEADER LINE.
gt_copy[] = gt_employee[].

WRITE: / '========== 9. MOVE/COPY OPERATION =========='.
WRITE: / 'Copied table contains', sy-tfill, 'records'.
LOOP AT gt_copy.
  WRITE: / gt_copy-emp_id, gt_copy-emp_name, gt_copy-emp_city.
ENDLOOP.
ULINE.

* 10. CLEAR OPERATION - Clearing header line only
CLEAR gt_employee.
WRITE: / '========== 10. CLEAR OPERATION (Header Line Only) =========='.
WRITE: / 'After CLEAR, header line is empty:'.
WRITE: / 'Header ID:', gt_employee-emp_id,
        'Header Name:', gt_employee-emp_name.
WRITE: / 'NOTE: Table body still contains data'.
DESCRIBE TABLE gt_employee LINES lv_lines.
WRITE: / 'Records still in table:', lv_lines.
ULINE.

* 11. REFRESH OPERATION - Clearing table body
REFRESH gt_employee.
WRITE: / '========== 11. REFRESH OPERATION (Clears Table Body) =========='.
DESCRIBE TABLE gt_employee LINES lv_lines.
WRITE: / 'Records after REFRESH:', lv_lines.
WRITE: / 'Table body is completely cleared.'.
WRITE: / 'But header line still has previous values:'.
WRITE: / 'Header ID:', gt_employee-emp_id.
ULINE.

* 12. FREE OPERATION - Releasing memory
FREE gt_employee.
WRITE: / '========== 12. FREE OPERATION (Releases Memory) =========='.
DESCRIBE TABLE gt_employee LINES lv_lines.
WRITE: / 'Records after FREE:', lv_lines.
WRITE: / 'Memory is completely released.'.
ULINE.

* 13. DESCRIBE OPERATION - Getting table properties
DATA: lv_lines_des TYPE i,
      lv_occurs    TYPE i,
      lv_kind      TYPE c,
      lv_key       TYPE i.

DESCRIBE TABLE gt_copy LINES lv_lines_des.
DESCRIBE TABLE gt_copy OCCURS lv_occurs.
DESCRIBE FIELD gt_copy TYPE lv_kind.

WRITE: / '========== 13. DESCRIBE OPERATION =========='.
WRITE: / 'Total lines in GT_COPY:', lv_lines_des.
WRITE: / 'Initial size (OCCURS):', lv_occurs.
WRITE: / 'Field type:', lv_kind.
ULINE.

* 14. ADDITIONAL OPERATIONS - Combined demo with backup table
WRITE: / '========== 14. ADDITIONAL COLLECT EXAMPLE (City-wise Sum) =========='.
DATA: lt_city_sum TYPE TABLE OF ty_employee WITH HEADER LINE.
REFRESH lt_city_sum.

LOOP AT gt_copy.
  lt_city_sum-emp_city = gt_copy-emp_city.
  lt_city_sum-emp_sal = gt_copy-emp_sal.
  COLLECT lt_city_sum.
ENDLOOP.

LOOP AT lt_city_sum.
  WRITE: / 'City:', lt_city_sum-emp_city,
          'Total Salary:', lt_city_sum-emp_sal.
ENDLOOP.
ULINE.

* 15. COMPLEX LOOP WITH CALCULATIONS
lv_total_sal = 0.
lv_count = 0.

LOOP AT gt_copy.
  lv_total_sal = lv_total_sal + gt_copy-emp_sal.
  lv_count = lv_count + 1.
ENDLOOP.

WRITE: / '========== 15. LOOP WITH CALCULATIONS =========='.
WRITE: / 'Total Employees:', lv_count.
IF lv_count > 0.
  WRITE: / 'Total Salary:', lv_total_sal.
ENDIF.
ULINE.

* 16. SORT AND DISPLAY
SORT gt_copy BY emp_sal DESCENDING.
WRITE: / '========== 16. SORTED TABLE (By Salary Descending) =========='.
WRITE: / 'Top 3 Highest Paid Employees:'.
lv_count = 0.
LOOP AT gt_copy.
  lv_count = lv_count + 1.
  WRITE: / gt_copy-emp_id, gt_copy-emp_name, gt_copy-emp_sal.
  IF lv_count >= 3.
    EXIT.
  ENDIF.
ENDLOOP.
ULINE.
