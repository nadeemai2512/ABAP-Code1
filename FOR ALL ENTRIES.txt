REPORT Y_FILL_FINALDATA.

* TYPES DECLARATION
TYPES : BEGIN OF MARA_TYPE,
          MATNR TYPE MATNR,
          MTART TYPE MTART,
          MBRSH TYPE MBRSH,
          MATKL TYPE MATKL,
          MEINS TYPE MEINS,
        END OF MARA_TYPE.

TYPES : BEGIN OF MARC_TYPE,
          MATNR TYPE MATNR,
          WERKS TYPE WERKS_D,
          EKGRP TYPE EKGRP,
          MINBE TYPE MINBE,
          BSTMI TYPE BSTMI,
        END OF MARC_TYPE.

TYPES : BEGIN OF FINAL_TYPE,
          MATNR TYPE MATNR,
          MTART TYPE MTART,
          MBRSH TYPE MBRSH,
          MATKL TYPE MATKL,
          MEINS TYPE MEINS,
          WERKS TYPE WERKS_D,
          EKGRP TYPE EKGRP,
          MINBE TYPE MINBE,
          BSTMI TYPE BSTMI,
        END OF FINAL_TYPE.
* DATA DECLARATION
DATA WA_MARA TYPE MARA_TYPE.
DATA ITAB_MARA TYPE TABLE OF MARA_TYPE.

DATA WA_MARC TYPE MARC_TYPE.
DATA ITAB_MARC TYPE TABLE OF MARC_TYPE.

DATA WA_FINAL TYPE FINAL_TYPE.
DATA ITAB_FINAL TYPE TABLE OF FINAL_TYPE.

DATA : N1 TYPE I,
       N2 TYPE I.

* START OF PROGRAM

SELECT MATNR MTART MBRSH MATKL MEINS
  FROM MARA
  INTO TABLE ITAB_MARA
  UP TO 200 ROWS
  WHERE MATNR BETWEEN '000000000000000010' AND '000000000000000500'.

IF ITAB_MARA IS NOT INITIAL.

  SELECT MATNR WERKS EKGRP MINBE BSTMI
    FROM MARC
    INTO TABLE ITAB_MARC
    FOR ALL ENTRIES IN ITAB_MARA
    WHERE MATNR = ITAB_MARA-MATNR.

  LOOP AT ITAB_MARC INTO WA_MARC.

    CLEAR WA_FINAL.

    WA_FINAL-WERKS = WA_MARC-WERKS.
    WA_FINAL-EKGRP = WA_MARC-EKGRP.
    WA_FINAL-MINBE = WA_MARC-MINBE.
    WA_FINAL-BSTMI = WA_MARC-BSTMI.

    READ TABLE ITAB_MARA INTO WA_MARA WITH KEY MATNR = WA_MARC-MATNR.

    IF SY-SUBRC = 0.
      WA_FINAL-MATNR = WA_MARA-MATNR.
      WA_FINAL-MTART = WA_MARA-MTART.
      WA_FINAL-MBRSH = WA_MARA-MBRSH.
      WA_FINAL-MATKL = WA_MARA-MATKL.
      WA_FINAL-MEINS = WA_MARA-MEINS.
    ENDIF.

    APPEND WA_FINAL TO ITAB_FINAL.

  ENDLOOP.

ENDIF.
* DISPLAY OUTPUT WITH COLORS

DESCRIBE TABLE ITAB_FINAL LINES N1 OCCURS N2.

FORMAT COLOR COL_HEADING INTENSIFIED ON.
WRITE: / '******************* MATERIAL FINAL REPORT *******************'.
FORMAT RESET.

SKIP.

FORMAT COLOR COL_TOTAL INTENSIFIED ON.
WRITE: / 'TOTAL RECORDS FOUND : ', N1.
FORMAT RESET.

SKIP.

* Header Line
FORMAT COLOR COL_GROUP INTENSIFIED ON.
WRITE: / 'MATNR', 20 'WERKS', 30 'EKGRP', 40 'MTART', 50 'MBRSH',
         60 'MATKL', 70 'MINBE', 85 'BSTMI', 100 'MEINS'.
FORMAT RESET.

ULINE.

LOOP AT ITAB_FINAL INTO WA_FINAL.

  FORMAT COLOR COL_NORMAL INTENSIFIED OFF.
  WRITE: / WA_FINAL-MATNR UNDER 'MATNR',
           WA_FINAL-WERKS UNDER 'WERKS',
           WA_FINAL-EKGRP UNDER 'EKGRP',
           WA_FINAL-MTART UNDER 'MTART',
           WA_FINAL-MBRSH UNDER 'MBRSH',
           WA_FINAL-MATKL UNDER 'MATKL',
           WA_FINAL-MINBE UNDER 'MINBE',
           WA_FINAL-BSTMI UNDER 'BSTMI',
           WA_FINAL-MEINS UNDER 'MEINS'.

ENDLOOP.

SKIP.

FORMAT COLOR COL_POSITIVE INTENSIFIED ON.
WRITE: / 'REPORT COMPLETED SUCCESSFULLY'.
FORMAT RESET.
