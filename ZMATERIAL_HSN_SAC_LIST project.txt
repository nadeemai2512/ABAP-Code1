*&      Form  FETCH_DATA
FORM FETCH_DATA .

  SELECT MATNR ERSDA ERNAM MTART MBRSH MATKL MEINS
    FROM MARA
    INTO TABLE IT_MARA
    WHERE MTART IN S_MTART.

  WRITE: / 'Materials found in MARA for given Material Type:', SY-DBCNT.

  IF IT_MARA IS NOT INITIAL.

    SELECT MATNR WERKS EKGRP STEUC
      FROM MARC
      INTO TABLE IT_MARC
      FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR
        AND WERKS IN S_WERKS
        AND STEUC IN S_STEUC.

    WRITE: / 'Materials found in MARC for given Plant/Control Code:', SY-DBCNT.

    SELECT MATNR SPRAS MAKTX
      FROM MAKT
      INTO TABLE IT_MAKT
      FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR
        AND SPRAS = SY-LANGU.

    SELECT MATNR BWKEY BWTAR BKLAS
      FROM MBEW
      INTO TABLE IT_MBEW
      FOR ALL ENTRIES IN IT_MARA
      WHERE MATNR = IT_MARA-MATNR.

  ELSE.
    WRITE: / 'No materials found in MARA. Check Material Type entry (must be uppercase, e.g. ROH, FERT, HALB, DIEN, VERP).'.
  ENDIF.

  IF IT_MARC IS NOT INITIAL.
    SELECT MATNR VKORG VTWEG DWERK KTGRM
      FROM MVKE
      INTO TABLE IT_MVKE
      FOR ALL ENTRIES IN IT_MARC
      WHERE MATNR = IT_MARC-MATNR
        AND DWERK = IT_MARC-WERKS.
  ELSE.
  ENDIF.

ENDFORM.
*&      Form  SORTING
FORM SORTING .
  SORT : IT_MARA BY MATNR,
         IT_MARC BY MATNR WERKS,
         IT_MAKT BY MATNR SPRAS,
         IT_MBEW BY MATNR BWKEY,
         IT_MVKE BY MATNR DWERK VKORG VTWEG.
ENDFORM.
*&      Form  PROCESS_DATA
FORM PROCESS_DATA .

  LOOP AT IT_MARC INTO WA_MARC.

    CLEAR WA_FINAL.

    WA_FINAL-MATNR = WA_MARC-MATNR.
    WA_FINAL-WERKS = WA_MARC-WERKS.
    WA_FINAL-EKGRP = WA_MARC-EKGRP.
    WA_FINAL-STEUC = WA_MARC-STEUC.

    READ TABLE IT_MARA INTO WA_MARA
         WITH KEY MATNR = WA_MARC-MATNR BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-ERSDA = WA_MARA-ERSDA.
      WA_FINAL-ERNAM = WA_MARA-ERNAM.
      WA_FINAL-MTART = WA_MARA-MTART.
      WA_FINAL-MBRSH = WA_MARA-MBRSH.
      WA_FINAL-MATKL = WA_MARA-MATKL.
      WA_FINAL-MEINS = WA_MARA-MEINS.
    ENDIF.

    READ TABLE IT_MAKT INTO WA_MAKT
         WITH KEY MATNR = WA_MARC-MATNR SPRAS = SY-LANGU BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-SPRAS = WA_MAKT-SPRAS.
      WA_FINAL-MAKTX = WA_MAKT-MAKTX.
    ENDIF.

    READ TABLE IT_MBEW INTO WA_MBEW
         WITH KEY MATNR = WA_MARC-MATNR BWKEY = WA_MARC-WERKS BINARY SEARCH.
    IF SY-SUBRC = 0.
      WA_FINAL-BWKEY = WA_MBEW-BWKEY.
      WA_FINAL-BWTAR = WA_MBEW-BWTAR.
      WA_FINAL-BKLAS = WA_MBEW-BKLAS.
    ENDIF.

    READ TABLE IT_MVKE INTO WA_MVKE
         WITH KEY MATNR = WA_MARC-MATNR DWERK = WA_MARC-WERKS BINARY SEARCH.
    IF SY-SUBRC <> 0.
*     Fallback: material's sales view may reference a different delivering plant.
*     Take the first sales-org entry found for this material instead of none at all.
      READ TABLE IT_MVKE INTO WA_MVKE WITH KEY MATNR = WA_MARC-MATNR BINARY SEARCH.
    ENDIF.
    IF SY-SUBRC = 0.
      WA_FINAL-VKORG = WA_MVKE-VKORG.
      WA_FINAL-VTWEG = WA_MVKE-VTWEG.
      WA_FINAL-DWERK = WA_MVKE-DWERK.
      WA_FINAL-KTGRM = WA_MVKE-KTGRM.
    ENDIF.

*   Mark genuinely unmaintained control/sales fields clearly instead of leaving blank
    IF WA_FINAL-STEUC IS INITIAL.
      WA_FINAL-STEUC = 'Not Maintained'.
    ENDIF.
    IF WA_FINAL-VKORG IS INITIAL.
      WA_FINAL-VKORG = '-'.
    ENDIF.
    IF WA_FINAL-VTWEG IS INITIAL.
      WA_FINAL-VTWEG = '-'.
    ENDIF.
    IF WA_FINAL-KTGRM IS INITIAL.
      WA_FINAL-KTGRM = '-'.
    ENDIF.

    APPEND WA_FINAL TO IT_FINAL.

    CLEAR : WA_FINAL, WA_MARA, WA_MARC, WA_MAKT, WA_MBEW, WA_MVKE.

  ENDLOOP.

ENDFORM.
*&      Form  BUILD_FCAT
FORM BUILD_FCAT .

  DEFINE MACRO_FCAT.
    CLEAR WA_FCAT.
    WA_FCAT-COL_POS   = &1.
    WA_FCAT-FIELDNAME = &2.
    WA_FCAT-JUST      = 'C'.
    WA_FCAT-TABNAME   = 'IT_FINAL'.
    WA_FCAT-SELTEXT_M = &3.
    APPEND WA_FCAT TO IT_FCAT.
  END-OF-DEFINITION.

  MACRO_FCAT 1  'MATNR' 'MATERIAL NUMBER'.
  MACRO_FCAT 2  'STEUC' 'CONTROL CODE (HSN/SAC)'.
  MACRO_FCAT 3  'MAKTX' 'MATERIAL DESCRIPTION'.
  MACRO_FCAT 4  'WERKS' 'PLANT'.
  MACRO_FCAT 5  'MTART' 'MATERIAL TYPE'.
  MACRO_FCAT 6  'MATKL' 'MATERIAL GROUP'.
  MACRO_FCAT 7  'MEINS' 'UNIT OF MEASURE'.
  MACRO_FCAT 8  'MBRSH' 'INDUSTRY SECTOR'.
  MACRO_FCAT 9  'EKGRP' 'PURCHASING GROUP'.
  MACRO_FCAT 10 'BWKEY' 'VALUATION AREA'.
  MACRO_FCAT 11 'BWTAR' 'VALUATION TYPE'.
  MACRO_FCAT 12 'BKLAS' 'VALUATION CLASS'.
  MACRO_FCAT 13 'VKORG' 'SALES ORG'.
  MACRO_FCAT 14 'VTWEG' 'DISTRIBUTION CHANNEL'.
  MACRO_FCAT 15 'DWERK' 'DELIVERY PLANT'.
  MACRO_FCAT 16 'KTGRM' 'ACCOUNT ASSGT GRP'.
  MACRO_FCAT 17 'ERNAM' 'CREATED BY'.
  MACRO_FCAT 18 'ERSDA' 'CREATED ON'.
  MACRO_FCAT 19 'SPRAS' 'LANGUAGE KEY'.

ENDFORM.
*&      Form  DISPLAY_OUTPUT
FORM DISPLAY_OUTPUT .

  IF IT_FINAL IS INITIAL.
    MESSAGE 'No data found for the given selection criteria' TYPE 'I'.
  ELSE.
    CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
      EXPORTING
        I_CALLBACK_PROGRAM = SY-REPID
        I_GRID_TITLE        = 'HSN & SAC CODES FOR MATERIALS'
        IT_FIELDCAT         = IT_FCAT
      TABLES
        T_OUTTAB             = IT_FINAL
      EXCEPTIONS
        PROGRAM_ERROR        = 1
        OTHERS                = 2.
    IF SY-SUBRC <> 0.
      MESSAGE 'Error displaying ALV output' TYPE 'E'.
    ENDIF.
  ENDIF.

ENDFORM.
